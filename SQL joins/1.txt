mysql> select d.dept_name,
    -> count(distinct e.emp_id) as total_employees,
    -> count(distinct p.project_id) as total_projects,
    -> coalesce(sum(distinct p.budget),0) as total_project_budget,
    -> coalesce(sum(distinct ep.hours_worked),0) as total_working_hours
    -> from department d
    -> left join employee e on d.dept_id=e.dept_id
    -> left join project p on d.dept_id=p.dept_id
    -> left join employee_project ep on e.emp_id=ep.emp_id and p.project_id=ep.project_id
    -> group by d.dept_id,d.dept_name;
